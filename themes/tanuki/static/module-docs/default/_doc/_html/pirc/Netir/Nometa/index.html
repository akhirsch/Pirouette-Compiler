  5. | CharTy of m

  1. | Neg of m

  3. | IntLit of m * int
  4. | FloatLit of m * float
  5. | CharLit of m * char

  8. | FalseLit of m

  10. | Match of m * expr * (pattern * expr) list
  11. | RecAbs of m * name * name * expr
  12. | FunApp of m * expr * expr
  13. | TypeConstr of m * expr * typ
  14. | Unop of m * unop * expr
  15. | Binop of m * binop * expr * expr
  16. | Send of m * expr * name
  17. | Recv of m * typ * name
  18. | ChooseFor of m * name * lab
  19. | AllowChoice of m * name * (lab * expr) list
  20. | AmI of m * expr
type decl =
  1. | EmulatedLocDecl of m * name
  2. | TypeDecl of m * name * typ
  3. | TypeAliasDecl of m * name * typ
  4. | DefnDecl of m * name * pattern list * expr
  5. | ImportDecl of m * name
